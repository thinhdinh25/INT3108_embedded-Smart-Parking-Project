\section{Phân công công việc}
Mỗi thành viên phụ trách chính một phân hệ, đồng thời cả nhóm cùng tham gia thiết kế
kiến trúc, tích hợp, kiểm thử và viết báo cáo. Công việc được chia để khối lượng của
ba thành viên tương đương nhau.

\begin{longtable}{@{}p{3.2cm}p{11.2cm}@{}}
\caption{Phân công công việc}\label{tab:phancong}\\
\toprule \textbf{Thành viên} & \textbf{Công việc chính} \\ \midrule \endfirsthead
\toprule \textbf{Thành viên} & \textbf{Công việc chính} \\ \midrule \endhead
Nguyễn Chí Minh\newline 24020232 &
\begin{itemize}[nosep,leftmargin=1.1em,before=\vspace{-\baselineskip}]
  \item Firmware máy chủ: kiến trúc phân lớp, tác vụ FreeRTOS, UDP/HTTP server, mDNS.
  \item Màn hình ST7789, trang web giám sát, nhật ký vào/ra.
  \item Tối ưu hiệu năng và bộ nhớ.
  \item Viết Chương 3, 5 của báo cáo.
\end{itemize} \\ \midrule
Đinh Đức Thịnh\newline 24020313 &
\begin{itemize}[nosep,leftmargin=1.1em,before=\vspace{-\baselineskip}]
  \item Firmware Sensor Client và RFID Client: cảm biến IR, lửa, nước; hai đầu đọc MFRC522, DHT11; OTA.
  \item Logic vào/ra theo cổng, điểm truy cập Wi-Fi, hiệu chỉnh chống rung.
  \item Lắp đặt, đấu dây cảm biến trên mô hình.
  \item Viết Chương 2, 4 của báo cáo.
\end{itemize} \\ \midrule
Nguyễn Việt Tiến\newline 24020322 &
\begin{itemize}[nosep,leftmargin=1.1em,before=\vspace{-\baselineskip}]
  \item Thanh chắn servo tự động, chế độ cháy, còi báo động.
  \item Công cụ giả lập, bộ kiểm thử, kịch bản kiểm thử phần cứng.
  \item Tài liệu hướng dẫn (README), mô hình bãi xe.
  \item Viết Chương 1, 6, 7, 8 của báo cáo.
\end{itemize} \\ \midrule
Cả nhóm &
\begin{itemize}[nosep,leftmargin=1.1em,before=\vspace{-\baselineskip}]
  \item Phân tích yêu cầu, thiết kế giao thức và kiến trúc tổng thể.
  \item Tích hợp ba bo mạch, chạy kịch bản demo, rà soát báo cáo.
\end{itemize} \\
\bottomrule
\end{longtable}
